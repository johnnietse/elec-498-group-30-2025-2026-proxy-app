# NOTE THAT FOR TEST B (for wraparound, so we can compare the energy with Test A and Test C properly):

BEFORE=$(cat /sys/class/powercap/intel-rapl:0/energy_uj)

mpirun -np 16 --bind-to core ./miniMD_openmpi -i in.lj.miniMD

AFTER=$(cat /sys/class/powercap/intel-rapl:0/energy_uj)
DIFF=$(( AFTER - BEFORE ))
if [ $DIFF -lt 0 ]; then
    DIFF=$(( DIFF + MAX_RANGE ))
fi
echo "Energy: $(( DIFF / 1000000 )) Joules"
